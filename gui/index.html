rzk/amortized_potential.rzk (Potential Method Invariant) 
#lang rzk-1

#doc
  Formalization of Physicist's Potential Method and Amortized Invariants.

#def DataStructureSpace : U := U
#def RealNonNeg : U := U

#def PotentialFunction ( D : DataStructureSpace ) : U := D -> RealNonNeg

#def AmortizedOperation
  ( D : DataStructureSpace )
  ( s0 s1 : D )
  ( actual_cost : RealNonNeg )
  ( phi : PotentialFunction D )
  : U
  := ( amortized_cost : RealNonNeg )
   * ( (t : 2) -> D [ t === 0_2 |-> s0, t === 1_2 |-> s1 ] )

#def AmortizedTelescopingProperty
  ( D : DataStructureSpace )
  ( phi : PotentialFunction D )
  ( s_init s_final : D )
  ( sum_actual : RealNonNeg )
  ( sum_amortized : RealNonNeg )
  : U
  := (sum_amortized >= sum_actual)
rzk/complexity_space.rzk (Directed Reduction 2-Simplices) 
#lang rzk-1

#doc
  Formalization of Computational Complexity State Spaces in Rzk.

#def ComputationSpace : U := U

#def ComputationStep
  ( C : ComputationSpace )
  ( s0 s1 : C )
  : U
  := (t : 2) -> C [ t === 0_2 |-> s0, t === 1_2 |-> s1 ]

#def StepComposition
  ( C : ComputationSpace )
  ( s0 s1 s2 : C )
  ( step01 : ComputationStep C s0 s1 )
  ( step12 : ComputationStep C s1 s2 )
  : U
  := (t1 : 2) -> (t2 : 2) -> (t1 <= t2) -> C [
       (t1 === 0_2) /\ (t2 === 0_2) |-> s0,
       (t1 === 0_2) /\ (t2 === 1_2) |-> s1,
       (t1 === 1_2) /\ (t2 === 1_2) |-> s2
     ]
